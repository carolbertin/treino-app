2.1 extractTokens(s)
Extrai tokens de uma string de prescrição independente de ordem ou formato.
function extractTokens(s) {
  const t = {};
  s = (s||&apos;&apos;).replace(/\s+/g, &apos; &apos;).trim();
  // Séries x reps (fixo ou range)
  const sr = s.match(/(\d+)\s*x\s*([\d]+-[\d]+|[\d]+)/i);
  if (sr) { t.series = +sr[1]; const rp = sr[2];
    const rg = rp.match(/(\d+)-(\d+)/);
    if (rg) { t.repsMin=+rg[1]; t.repsMax=+rg[2]; t.repsStr=rg[1]+&apos;–&apos;+rg[2]; }
    else { t.repsMin=t.repsMax=+rp; t.repsStr=rp; } }
  // Cluster blocks: 4+4+4
  const cb = s.match(/(\d+)\+(\d+)\+(\d+)(?:\+(\d+))?/);
  if (cb) { t.cluster=[cb[1],cb[2],cb[3]]; if(cb[4])t.cluster.push(cb[4]); }
  // RIR
  const rir = s.match(/\((\d+)\s*rir\)/i); if(rir) t.rir=+rir[1];
  // Drop % 
  const pct = s.match(/-\s*(\d+)\s*%/); if(pct) t.dropPct=+pct[1];
  // Rest seconds
  const rest = s.match(/(\d+)\s*seg/i); if(rest) t.restSec=+rest[1];
  // Cluster rest
  const cr = s.match(/(\d+)(?:\/(\d+))?\s*seg\s*entre/i);
  if (cr) t.clusterRest = cr[1]+(cr[2]?&apos;–&apos;+cr[2]:&apos;&apos;)+&apos;s&apos;;
  // Continuation range +4/5 or +99
  const cRange = s.match(/\+\s*(\d+)\s*\/\s*(\d+)/);
  if (cRange) { t.contMin=+cRange[1]; t.contMax=+cRange[2]; t.contStr=cRange[1]+&apos;/&apos;+cRange[2]; }
  else { const c99=s.match(/\+\s*99/); if(c99) t.cont99=true;
    else { const cn=s.match(/\+\s*(\d+)/); if(cn){t.contMin=t.contMax=+cn[1];t.contStr=cn[1];} } }
  return t;
}

Adicionar ao JS. A função iniciarTimer() existente continua funcionando para usos legados.
let restTimerRAF = null;
let restTimerStart = null;
let restTimerCallback = null;
function startRestTimer(secs, onComplete) {
  restTimerCallback = onComplete;
  restTimerStart = Date.now();
  $(&apos;rest-timer-num&apos;).textContent = secs;
  $(&apos;rest-timer-overlay&apos;).style.display = &apos;flex&apos;;
  function tick() {
    const elapsed = Math.floor((Date.now()-restTimerStart)/1000);
    const rem = Math.max(0, secs-elapsed);
    $(&apos;rest-timer-num&apos;).textContent = rem;
    $(&apos;rest-timer-circle&apos;).style.strokeDashoffset = (326.7*elapsed/secs).toFixed(1);
    if (rem <= 0) {
      $(&apos;rest-timer-overlay&apos;).style.display = &apos;none&apos;;
      if (navigator.vibrate) navigator.vibrate(200);
      setTimeout(() => { if(restTimerCallback) restTimerCallback(); }, 300);
      return;
    }
    restTimerRAF = requestAnimationFrame(tick);
  }
  restTimerRAF = requestAnimationFrame(tick);
}
function pularRestTimer() {
  if(restTimerRAF) cancelAnimationFrame(restTimerRAF);
  $(&apos;rest-timer-overlay&apos;).style.display = &apos;none&apos;;
  if(restTimerCallback) restTimerCallback();
  restTimerCallback = null;
}

Critério de conclusão: Ao finalizar uma série, o overlay fullscreen aparece com countdown animado. &apos;Pular descanso&apos; avança imediatamente. App continua salvando normalmente.
